```python
# arc_see/lenses/__init__.py
"""Lens system — pure functions that transform grids into annotations."""

from __future__ import annotations

import numpy as np
from dataclasses import dataclass, field
from typing import Protocol, runtime_checkable


# --- Annotation types ---

@dataclass(frozen=True)
class RegionLabels:
    """Output of spatial lenses: integer-labeled regions."""
    labels: np.ndarray          # (64, 64) int array, 0 = background
    count: int                  # number of distinct regions (excluding background)

@dataclass(frozen=True)
class EdgeMask:
    """Binary edge mask."""
    mask: np.ndarray            # (64, 64) bool array

@dataclass(frozen=True)
class ContainmentTree:
    """Nested containment: which regions are inside which."""
    parent: dict[int, int | None]   # region_id → parent_region_id (None = top-level)
    children: dict[int, list[int]]  # region_id → list of contained region_ids

@dataclass(frozen=True)
class DisplacementField:
    """Per-object displacement vectors between frames."""
    vectors: dict[int, tuple[float, float]]  # object_id → (dy, dx)

@dataclass(frozen=True)
class ChangeMask:
    """Appeared/disappeared or same-position-color-change mask."""
    appeared: np.ndarray        # (64, 64) bool
    disappeared: np.ndarray     # (64, 64) bool

@dataclass(frozen=True)
class StateChangeMask:
    """Same position, different color."""
    mask: np.ndarray            # (64, 64) bool
    old_colors: np.ndarray      # (64, 64) int, valid where mask is True
    new_colors: np.ndarray      # (64, 64) int, valid where mask is True

@dataclass(frozen=True)
class SymmetryResult:
    """Detected symmetry axes."""
    horizontal: bool
    vertical: bool
    rotational_order: int       # 0 = none, 2 = 180°, 4 = 90°

@dataclass(frozen=True)
class PeriodicityResult:
    """Detected repeating patterns."""
    row_period: int | None      # None if no row periodicity
    col_period: int | None

@dataclass(frozen=True)
class AlignmentResult:
    """Grid-like arrangement detection."""
    row_lines: list[int]        # y-coordinates of detected row boundaries
    col_lines: list[int]        # x-coordinates of detected column boundaries

@dataclass(frozen=True)
class ProximityGraph:
    """Distance graph between objects."""
    distances: dict[tuple[int, int], float]  # (obj_a, obj_b) → centroid distance

@dataclass(frozen=True)
class ColorCorrespondence:
    """Color-to-object-group mapping."""
    color_to_objects: dict[int, list[int]]   # color → list of object_ids
    object_to_color: dict[int, int]          # object_id → dominant color

@dataclass(frozen=True)
class SizeCount:
    """Object sizes and group counts."""
    sizes: dict[int, int]       # object_id → pixel count
    count_by_size: dict[int, int]  # pixel_count → number of objects with that size

@dataclass(frozen=True)
class GoalDetection:
    """Candidate goal pattern and similarity to current state."""
    pattern: np.ndarray | None  # extracted goal region, or None if no goal detected
    location: tuple[int, int, int, int] | None  # (y1, x1, y2, x2) bounding box
    similarity: float           # 0.0-1.0, similarity of current state to goal


# Union of all annotation types
Annotation = (
    RegionLabels | EdgeMask | ContainmentTree |
    DisplacementField | ChangeMask | StateChangeMask |
    SymmetryResult | PeriodicityResult | AlignmentResult |
    ProximityGraph | ColorCorrespondence | SizeCount |
    GoalDetection
)


# --- Lens protocols ---

@runtime_checkable
class Lens(Protocol):
    """A lens that operates on a single grid."""
    name: str
    def __call__(self, grid: np.ndarray) -> Annotation: ...

@runtime_checkable
class TemporalLens(Protocol):
    """A lens that operates on grid history."""
    name: str
    def __call__(self, current: np.ndarray, history: list[np.ndarray]) -> Annotation: ...

@runtime_checkable
class DependentLens(Protocol):
    """A lens that depends on another lens's output."""
    name: str
    depends_on: str  # name of the lens this depends on
    def __call__(self, grid: np.ndarray, dependency: Annotation) -> Annotation: ...


# --- Registry ---

_LENSES: dict[str, Lens | TemporalLens | DependentLens] = {}

def register(lens: Lens | TemporalLens | DependentLens) -> None:
    _LENSES[lens.name] = lens

def get_all() -> dict[str, Lens | TemporalLens | DependentLens]:
    return dict(_LENSES)
```

```python
# arc_see/lenses/spatial.py
"""Spatial lenses: what's on the grid?"""

from __future__ import annotations

import numpy as np
from scipy import ndimage

from arc_see.lenses import RegionLabels, EdgeMask, ContainmentTree, register


class _ConnectedComponents:
    name = "connected_components"

    def __call__(self, grid: np.ndarray) -> RegionLabels:
        # Treat 0 as background — label each contiguous non-zero region
        mask = grid != 0
        # Label connected regions, treating different colors as different objects
        # by labeling each color separately then merging
        labels = np.zeros_like(grid, dtype=np.int32)
        next_label = 1
        for color in range(1, 16):
            color_mask = grid == color
            if not color_mask.any():
                continue
            color_labels, n = ndimage.label(color_mask)
            for i in range(1, n + 1):
                labels[color_labels == i] = next_label
                next_label += 1
        return RegionLabels(labels=labels, count=next_label - 1)


connected_components = _ConnectedComponents()
register(connected_components)
```

```python
class _Edges:
    name = "edges"

    def __call__(self, grid: np.ndarray) -> EdgeMask:
        # A cell is an edge if any of its 4-neighbors has a different color
        padded = np.pad(grid, 1, mode='edge')
        up = padded[:-2, 1:-1]
        down = padded[2:, 1:-1]
        left = padded[1:-1, :-2]
        right = padded[1:-1, 2:]
        mask = (
            (grid != up) | (grid != down) |
            (grid != left) | (grid != right)
        )
        return EdgeMask(mask=mask)


edges = _Edges()
register(edges)
```

```python
# Add to arc_see/lenses/spatial.py

class _Enclosure:
    name = "enclosure"
    depends_on = "connected_components"

    def __call__(self, grid: np.ndarray, dependency: RegionLabels) -> ContainmentTree:
        labels = dependency.labels
        parent: dict[int, int | None] = {}
        children: dict[int, list[int]] = {}

        # For each region, find its bounding box
        bboxes: dict[int, tuple[int, int, int, int]] = {}
        for region_id in range(1, dependency.count + 1):
            ys, xs = np.where(labels == region_id)
            if len(ys) == 0:
                continue
            bboxes[region_id] = (ys.min(), xs.min(), ys.max(), xs.max())

        region_ids = list(bboxes.keys())

        # Sort by area descending — larger regions checked first as potential parents
        region_ids.sort(key=lambda r: (
            (bboxes[r][2] - bboxes[r][0]) * (bboxes[r][3] - bboxes[r][1])
        ), reverse=True)

        for region_id in region_ids:
            parent[region_id] = None
            children.setdefault(region_id, [])

        # For each region, find the smallest region whose bbox fully contains it
        for i, rid in enumerate(region_ids):
            y1, x1, y2, x2 = bboxes[rid]
            best_parent = None
            best_area = float('inf')
            for j, candidate in enumerate(region_ids):
                if candidate == rid:
                    continue
                cy1, cx1, cy2, cx2 = bboxes[candidate]
                c_area = (cy2 - cy1) * (cx2 - cx1)
                if cy1 <= y1 and cx1 <= x1 and cy2 >= y2 and cx2 >= x2:
                    if c_area < best_area:
                        best_area = c_area
                        best_parent = candidate
            if best_parent is not None:
                parent[rid] = best_parent
                children[best_parent].append(rid)

        return ContainmentTree(parent=parent, children=children)


enclosure = _Enclosure()
register(enclosure)
```

```python
# arc_see/lenses/temporal.py
"""Temporal lenses: what changed between frames?"""

from __future__ import annotations

import numpy as np
from scipy import ndimage

from arc_see.lenses import (
    DisplacementField, ChangeMask, StateChangeMask,
    RegionLabels, register,
)


class _Movement:
    name = "movement"
    depends_on = "connected_components"

    def __call__(
        self,
        current: np.ndarray,
        history: list[np.ndarray],
        dependency: RegionLabels,
    ) -> DisplacementField:
        if not history:
            return DisplacementField(vectors={})

        prev = history[-1]
        prev_labels, prev_n = self._label_grid(prev)
        curr_labels, curr_n = self._label_grid(current)

        # Match objects by color and find centroid displacement
        vectors: dict[int, tuple[float, float]] = {}
        prev_centroids = self._centroids(prev, prev_labels, prev_n)
        curr_centroids = self._centroids(current, curr_labels, curr_n)

        # Match by color — for each color present in both frames,
        # pair objects by nearest centroid
        for color in range(1, 16):
            prev_objs = [(oid, c) for oid, (c, col) in prev_centroids.items() if col == color]
            curr_objs = [(oid, c) for oid, (c, col) in curr_centroids.items() if col == color]
            if not prev_objs or not curr_objs:
                continue
            # Simple nearest-neighbor matching
            for curr_id, curr_c in curr_objs:
                best_dist = float('inf')
                best_prev_c = None
                for prev_id, prev_c in prev_objs:
                    dist = (curr_c[0] - prev_c[0])**2 + (curr_c[1] - prev_c[1])**2
                    if dist < best_dist:
                        best_dist = dist
                        best_prev_c = prev_c
                if best_prev_c is not None:
                    dy = curr_c[0] - best_prev_c[0]
                    dx = curr_c[1] - best_prev_c[1]
                    vectors[curr_id] = (dy, dx)

        return DisplacementField(vectors=vectors)

    def _label_grid(self, grid: np.ndarray) -> tuple[np.ndarray, int]:
        labels = np.zeros_like(grid, dtype=np.int32)
        n = 0
        for color in range(1, 16):
            color_mask = grid == color
            if not color_mask.any():
                continue
            color_labels, count = ndimage.label(color_mask)
            for i in range(1, count + 1):
                n += 1
                labels[color_labels == i] = n
        return labels, n

    def _centroids(
        self, grid: np.ndarray, labels: np.ndarray, n: int
    ) -> dict[int, tuple[tuple[float, float], int]]:
        """Returns {label: ((cy, cx), color)}."""
        result = {}
        for obj_id in range(1, n + 1):
            ys, xs = np.where(labels == obj_id)
            if len(ys) == 0:
                continue
            color = grid[ys[0], xs[0]]
            result[obj_id] = ((ys.mean(), xs.mean()), color)
        return result


movement = _Movement()
register(movement)
```

```python
        new_colors = np.where(changed, current, 0)
```

```python
    def __call__(self, grid: np.ndarray) -> SymmetryResult:

        horizontal = np.array_equal(grid, grid[::-1, :])
        # Vertical symmetry: left half mirrors right half
        vertical = np.array_equal(grid, grid[:, ::-1])

        # 90° rotational symmetry

            rotational_order = 4

            rotational_order = 2
        else:
            rotational_order = 0
```

```python
class _Periodicity:
    name = "periodicity"

    def __call__(self, grid: np.ndarray) -> PeriodicityResult:
        row_period = self._find_period_along_axis(grid, axis=0)
        col_period = self._find_period_along_axis(grid, axis=1)
        return PeriodicityResult(row_period=row_period, col_period=col_period)

    def _find_period_along_axis(self, grid: np.ndarray, axis: int) -> int | None:
        """Check if the grid repeats along the given axis."""
        size = grid.shape[axis]
        for period in range(2, size // 2 + 1):
            if size % period != 0:
                continue
            # Check if grid[0:period] == grid[period:2*period] == ...
            if axis == 0:
                tile = grid[:period, :]
                matches = all(
                    np.array_equal(tile, grid[i:i+period, :])
                    for i in range(period, size, period)
                )
            else:
                tile = grid[:, :period]
                matches = all(
                    np.array_equal(tile, grid[:, i:i+period])
                    for i in range(period, size, period)
                )
            if matches:
                return period
        return None


periodicity = _Periodicity()
```

```python
class _Alignment:
    name = "alignment"

    def __call__(self, grid: np.ndarray) -> AlignmentResult:
        h, w = grid.shape
        row_lines = self._find_dividers(grid, axis=0)
        col_lines = self._find_dividers(grid, axis=1)
        return AlignmentResult(row_lines=row_lines, col_lines=col_lines)

    def _find_dividers(self, grid: np.ndarray, axis: int) -> list[int]:
        """Find lines where the grid content changes uniformly.

        A divider row/col is one where the content is uniform (all same color)
        and different from the rows/cols on either side, OR where there's a
        consistent transition in the perpendicular content.
        """
        size = grid.shape[axis]
        dividers = []

        # Approach: look for rows/cols that are entirely background (color 0)
        # bordered by non-background on at least one side
        for i in range(1, size - 1):
            if axis == 0:
                line = grid[i, :]
                prev_line = grid[i-1, :]
                next_line = grid[i+1, :]
            else:
                line = grid[:, i]
                prev_line = grid[:, i-1]
                next_line = grid[:, i+1]

            line_uniform = np.all(line == line[0])
            if line_uniform:
                prev_has_content = np.any(prev_line != line[0])
                next_has_content = np.any(next_line != line[0])
                if prev_has_content and next_has_content:
                    dividers.append(i)

        return dividers
```

```python
from __future__ import annotations

import numpy as np
from math import sqrt

from arc_see.lenses import (
    ProximityGraph, ColorCorrespondence, SizeCount,
    RegionLabels, register,
)


class _Proximity:
    name = "proximity"
    depends_on = "connected_components"

    def __call__(self, grid: np.ndarray, dependency: RegionLabels) -> ProximityGraph:
        labels = dependency.labels
        centroids: dict[int, tuple[float, float]] = {}
        for obj_id in range(1, dependency.count + 1):
            ys, xs = np.where(labels == obj_id)
            if len(ys) == 0:
                continue
            centroids[obj_id] = (ys.mean(), xs.mean())

        distances: dict[tuple[int, int], float] = {}
        obj_ids = sorted(centroids.keys())
        for i, a in enumerate(obj_ids):
            for b in obj_ids[i+1:]:
                cy_a, cx_a = centroids[a]
                cy_b, cx_b = centroids[b]
                dist = sqrt((cy_a - cy_b)**2 + (cx_a - cx_b)**2)
                distances[(a, b)] = dist

        return ProximityGraph(distances=distances)


proximity = _Proximity()
register(proximity)
```

```python
    name = "color_correspondence"
    depends_on = "connected_components"

    def __call__(self, grid: np.ndarray, dependency: RegionLabels) -> ColorCorrespondence:
        labels = dependency.labels
        color_to_objects: dict[int, list[int]] = {}
        object_to_color: dict[int, int] = {}

        for obj_id in range(1, dependency.count + 1):
            ys, xs = np.where(labels == obj_id)
            if len(ys) == 0:
                continue
            # Dominant color = most common non-zero color in the region
            colors = grid[ys, xs]
            nonzero = colors[colors != 0]
            if len(nonzero) == 0:
                continue
            values, counts = np.unique(nonzero, return_counts=True)
            dominant = values[counts.argmax()]
            object_to_color[obj_id] = int(dominant)
            color_to_objects.setdefault(int(dominant), []).append(obj_id)

        return ColorCorrespondence(
            color_to_objects=color_to_objects,
            object_to_color=object_to_color,
        )
```

```python
# arc_see/lenses/goal.py
"""Goal lens: detect candidate goal patterns in grid corners/edges."""

from __future__ import annotations

import numpy as np
from scipy import ndimage

from arc_see.lenses import GoalDetection, register


# Corner/edge regions to scan for goal indicators
_CORNER_REGIONS = [
    (0, 0, 8, 8),          # top-left
    (0, 56, 8, 64),        # top-right
    (56, 0, 64, 8),        # bottom-left
    (56, 56, 64, 64),      # bottom-right
]

_EDGE_REGIONS = [
    (0, 0, 4, 64),         # top edge
    (60, 0, 64, 64),       # bottom edge
    (0, 0, 64, 4),         # left edge
    (0, 60, 64, 64),       # right edge
]


class _GoalDetection:
    name = "goal_detection"

    def __call__(self, grid: np.ndarray) -> GoalDetection:
        # Scan corner and edge regions for isolated small patterns
        best_candidate = None
        best_score = 0.0
        best_location = None

        for y1, x1, y2, x2 in _CORNER_REGIONS + _EDGE_REGIONS:
            region = grid[y1:y2, x1:x2]
            filled = region != 0
            fill_count = filled.sum()

            if fill_count == 0 or fill_count > 64:
                # No content or too much content to be a goal indicator
                continue

            # Check that the content is isolated (not connected to main grid)
            # by checking if the filled pixels touch the region boundary
            # that faces the interior of the grid
            content_ratio = fill_count / region.size
            if content_ratio > 0.5:
                continue

            # Score: prefer small, isolated, corner-located patterns
            isolation_score = 1.0 - content_ratio
            corner_bonus = 1.5 if (y1, x1, y2, x2) in _CORNER_REGIONS else 1.0
            score = isolation_score * corner_bonus

            if score > best_score:
                best_score = score
                # Extract just the filled portion
                ys, xs = np.where(filled)
                if len(ys) == 0:
                    continue
                py1, py2 = ys.min(), ys.max() + 1
                px1, px2 = xs.min(), xs.max() + 1
                best_candidate = region[py1:py2, px1:px2].copy()
                best_location = (y1 + py1, x1 + px1, y1 + py2, x1 + px2)

        if best_candidate is None:
            return GoalDetection(pattern=None, location=None, similarity=0.0)

        # Compute similarity: how well does the rest of the grid match the goal?
        similarity = self._compute_similarity(grid, best_candidate, best_location)

        return GoalDetection(
            pattern=best_candidate,
            location=best_location,
            similarity=similarity,
        )

    def _compute_similarity(
        self,
        grid: np.ndarray,
        pattern: np.ndarray,
        location: tuple[int, int, int, int],
    ) -> float:
        """Compute how similar the grid content (excluding goal region) is to the pattern."""
        y1, x1, y2, x2 = location
        ph, pw = pattern.shape

        # Mask out the goal region
        work = grid.copy()

        if not work.any():

        grid_colors = set(work[work != 0].flatten().tolist())

        if not pattern_colors or not grid_colors:
            return 0.0

        # Jaccard similarity of color sets
        intersection = len(pattern_colors & grid_colors)
        union = len(pattern_colors | grid_colors)
        color_sim = intersection / union if union > 0 else 0.0

        # Size similarity: does the grid have a region of similar size?
        pattern_size = np.sum(pattern != 0)
        labels, n = ndimage.label(work != 0)
        if n == 0:
            return color_sim * 0.5

        sizes = [np.sum(labels == i) for i in range(1, n + 1)]
        closest_size = min(sizes, key=lambda s: abs(s - pattern_size))
        size_ratio = min(pattern_size, closest_size) / max(pattern_size, closest_size)

        return (color_sim + size_ratio) / 2.0
```

```python
# arc_see/composer.py
"""Composer: run lenses, score combinations, surface top-K."""

from __future__ import annotations

import numpy as np
from itertools import combinations

from arc_see.lenses import (
    Annotation, Lens, TemporalLens, DependentLens,
    RegionLabels, EdgeMask, ContainmentTree,
    DisplacementField, ChangeMask, StateChangeMask,
    SymmetryResult, PeriodicityResult, AlignmentResult,
    ProximityGraph, ColorCorrespondence, SizeCount,
    GoalDetection,
    get_all,
)


class Composer:
    def perceive(
        self,
        grid: np.ndarray,
        history: list[np.ndarray],
    ) -> dict[str, Annotation]:
        """Run all lenses and return their annotations."""
        all_lenses = get_all()
        results: dict[str, Annotation] = {}

        # Phase 1: independent lenses (Lens protocol — single grid)
        for name, lens in all_lenses.items():
            if isinstance(lens, DependentLens) or isinstance(lens, TemporalLens):
                continue
            if hasattr(lens, 'depends_on'):
                continue
            # Check if it's a temporal lens by inspecting call signature
            try:
                results[name] = lens(grid)
            except TypeError:
                pass

        # Phase 2: temporal lenses (need history)
        for name, lens in all_lenses.items():
            if not hasattr(lens, '__call__'):
                continue
            # Temporal lenses have 'history' in their signature
            import inspect
            sig = inspect.signature(lens.__call__)
            params = list(sig.parameters.keys())
            if 'history' in params and 'dependency' not in params:
                results[name] = lens(grid, history)
            elif 'history' in params and 'dependency' in params:
                dep_name = lens.depends_on
                if dep_name in results:
                    results[name] = lens(grid, history, results[dep_name])

        # Phase 3: dependent lenses (need another lens's output)
        for name, lens in all_lenses.items():
            if name in results:
                continue
            if hasattr(lens, 'depends_on'):
                dep_name = lens.depends_on
                if dep_name in results:
                    try:
                        results[name] = lens(grid, results[dep_name])
                    except TypeError:
                        pass

        return results

    def score_combinations(
        self, annotations: dict[str, Annotation]
    ) -> list[tuple[float, tuple[str, str]]]:
        """Score all pairwise combinations by information content."""
        names = sorted(annotations.keys())
        scored: list[tuple[float, tuple[str, str]]] = []

        for a, b in combinations(names, 2):
            score = self._mutual_information(annotations[a], annotations[b])
            scored.append((score, (a, b)))

        scored.sort(key=lambda x: x[0], reverse=True)
        return scored

    def top_k(
        self,
        annotations: dict[str, Annotation],
        k: int = 5,
    ) -> list[tuple[float, tuple[str, str]]]:
        """Return the top-K most informative pairwise combinations."""
        scored = self.score_combinations(annotations)
        return scored[:k]

    def _mutual_information(self, a: Annotation, b: Annotation) -> float:
        """Estimate mutual information between two annotations.

        Higher score = the combination reveals more structure than either alone.
        Uses type-specific heuristics rather than formal entropy computation.
        """
        score = 0.0

        # Region + Color: how well do colors predict regions?
        if isinstance(a, RegionLabels) and isinstance(b, ColorCorrespondence):
            score = self._region_color_mi(a, b)
        elif isinstance(b, RegionLabels) and isinstance(a, ColorCorrespondence):
            score = self._region_color_mi(b, a)

        # Region + Proximity: spatial clustering quality
        elif isinstance(a, RegionLabels) and isinstance(b, ProximityGraph):
            score = self._region_proximity_mi(a, b)
        elif isinstance(b, RegionLabels) and isinstance(a, ProximityGraph):
            score = self._region_proximity_mi(b, a)

        # Region + Size: size uniformity reveals structure
        elif isinstance(a, RegionLabels) and isinstance(b, SizeCount):
            score = self._region_size_mi(a, b)
        elif isinstance(b, RegionLabels) and isinstance(a, SizeCount):
            score = self._region_size_mi(b, a)

        # Symmetry + Alignment: symmetric grid = highly structured
        elif isinstance(a, SymmetryResult) and isinstance(b, AlignmentResult):
            score = self._symmetry_alignment_mi(a, b)
        elif isinstance(b, SymmetryResult) and isinstance(a, AlignmentResult):
            score = self._symmetry_alignment_mi(b, a)

        # Movement + Goal: movement toward goal = high information
        elif isinstance(a, DisplacementField) and isinstance(b, GoalDetection):
            score = self._movement_goal_mi(a, b)
        elif isinstance(b, DisplacementField) and isinstance(a, GoalDetection):
            score = self._movement_goal_mi(b, a)

        # Edge + Enclosure: edges that form containers
        elif isinstance(a, EdgeMask) and isinstance(b, ContainmentTree):
            score = self._edge_enclosure_mi(a, b)
        elif isinstance(b, EdgeMask) and isinstance(a, ContainmentTree):
            score = self._edge_enclosure_mi(b, a)

        # Default: feature count overlap heuristic
        else:
            score = self._default_mi(a, b)

        return score

    def _region_color_mi(self, regions: RegionLabels, colors: ColorCorrespondence) -> float:
        """1:1 color→region mapping = high MI."""
        if regions.count == 0:
            return 0.0
        n_colors = len(colors.color_to_objects)
        if n_colors == 0:
            return 0.0
        # Perfect correspondence: each color maps to exactly one region
        perfect = all(len(objs) == 1 for objs in colors.color_to_objects.values())
        ratio = min(n_colors, regions.count) / max(n_colors, regions.count)
        return (1.0 if perfect else 0.5) * ratio

    def _region_proximity_mi(self, regions: RegionLabels, prox: ProximityGraph) -> float:
        """Clustered distances = high MI."""
        if not prox.distances:
            return 0.0
        dists = list(prox.distances.values())
        if len(dists) < 2:
            return 0.3
        mean_d = np.mean(dists)
        std_d = np.std(dists)
        # Low variance in distances → regular spacing → structure
        cv = std_d / mean_d if mean_d > 0 else 0
        return max(0, 1.0 - cv)

    def _region_size_mi(self, regions: RegionLabels, sizes: SizeCount) -> float:
        """Uniform sizes = grid of similar objects = high MI."""
        if not sizes.count_by_size:
            return 0.0
        # If most objects are the same size, that's structured
        max_count = max(sizes.count_by_size.values())
        total = sum(sizes.count_by_size.values())
        return max_count / total

    def _symmetry_alignment_mi(self, sym: SymmetryResult, align: AlignmentResult) -> float:
        """Symmetric + grid-aligned = highly structured."""
        sym_score = (sym.horizontal + sym.vertical + (sym.rotational_order > 0)) / 3.0
        align_score = min(len(align.row_lines) + len(align.col_lines), 10) / 10.0
        return sym_score * align_score

    def _movement_goal_mi(self, mov: DisplacementField, goal: GoalDetection) -> float:
        """Movement toward a detected goal = very informative."""
        if goal.pattern is None or not mov.vectors:
            return 0.0
        return goal.similarity * 0.8  # Strong signal when both active

    def _edge_enclosure_mi(self, edge: EdgeMask, enc: ContainmentTree) -> float:
        """Edges that form containment = structural."""
        n_nested = sum(1 for p in enc.parent.values() if p is not None)
        total = len(enc.parent)
        if total == 0:
            return 0.0
        return n_nested / total

    def _default_mi(self, a: Annotation, b: Annotation) -> float:
        """Fallback: small positive score for any two non-trivial annotations."""
        a_has_content = self._has_content(a)
        b_has_content = self._has_content(b)
        if a_has_content and b_has_content:
            return 0.1
        return 0.0

    def _has_content(self, ann: Annotation) -> bool:
        """Check if an annotation found anything non-trivial."""
        if isinstance(ann, RegionLabels):
            return ann.count > 0
        if isinstance(ann, EdgeMask):
            return ann.mask.any()
        if isinstance(ann, ContainmentTree):
            return any(p is not None for p in ann.parent.values())
        if isinstance(ann, DisplacementField):
            return len(ann.vectors) > 0
        if isinstance(ann, ChangeMask):
            return ann.appeared.any() or ann.disappeared.any()
        if isinstance(ann, StateChangeMask):
            return ann.mask.any()
        if isinstance(ann, SymmetryResult):
            return ann.horizontal or ann.vertical or ann.rotational_order > 0
        if isinstance(ann, PeriodicityResult):
            return ann.row_period is not None or ann.col_period is not None
        if isinstance(ann, AlignmentResult):
            return len(ann.row_lines) > 0 or len(ann.col_lines) > 0
        if isinstance(ann, ProximityGraph):
            return len(ann.distances) > 0
        if isinstance(ann, ColorCorrespondence):
            return len(ann.color_to_objects) > 0
        if isinstance(ann, SizeCount):
            return len(ann.sizes) > 0
        if isinstance(ann, GoalDetection):
            return ann.pattern is not None
        return False
```

```python
# arc_see/graph.py
"""In-game memory graph: state → action → state transitions."""

from __future__ import annotations

import numpy as np
from dataclasses import dataclass


@dataclass
class Transition:
    state_vec: np.ndarray
    action: str
    next_state_vec: np.ndarray
    state_changed: bool
    goal_delta: float
    active_lenses: list[tuple[str, ...]]
    turn: int


@dataclass
class QueryResult:
    action: str
    score: float
    state_changed: bool
    goal_delta: float
    similarity: float


class MemoryGraph:
    def __init__(self, similarity_threshold: float = 0.7, confidence_threshold: int = 5):
        self._transitions: list[Transition] = []
        self._turn: int = 0
        self._similarity_threshold = similarity_threshold
        self._confidence_threshold = confidence_threshold

    @property
    def size(self) -> int:
        return len(self._transitions)

    def record(
        self,
        state_vec: np.ndarray,
        action: str,
        next_state_vec: np.ndarray,
        state_changed: bool,
        goal_delta: float,
        active_lenses: list[tuple[str, ...]],
    ) -> None:
        self._transitions.append(Transition(
            state_vec=state_vec,
            action=action,
            next_state_vec=next_state_vec,
            state_changed=state_changed,
            goal_delta=goal_delta,
            active_lenses=active_lenses,
            turn=self._turn,
        ))
        self._turn += 1

    def query(
        self,
        state_vec: np.ndarray,
        max_results: int = 10,
    ) -> list[QueryResult]:
        """Find transitions from similar perceived states, ranked by utility."""
        if not self._transitions:
            return []

        candidates: list[QueryResult] = []
        for t in self._transitions:
            sim = self._cosine_similarity(state_vec, t.state_vec)
            if sim < self._similarity_threshold:
                continue

            # Score: goal_delta (primary) + novelty + recency
            goal_score = t.goal_delta  # can be negative
            novelty_score = 1.0 if t.state_changed else 0.0
            recency_score = t.turn / max(self._turn, 1)  # 0..1, higher = more recent

            score = (
                goal_score * 3.0 +
                novelty_score * 2.0 +
                recency_score * 0.5 +
                sim * 0.5
            )

            candidates.append(QueryResult(
                action=t.action,
                score=score,
                state_changed=t.state_changed,
                goal_delta=t.goal_delta,
                similarity=sim,
            ))

        candidates.sort(key=lambda r: r.score, reverse=True)
        return candidates[:max_results]

    def confidence(self, state_vec: np.ndarray) -> float:
        """How much data do we have for states like this? 0..1."""
        if not self._transitions:
            return 0.0
        matches = sum(
            1 for t in self._transitions
            if self._cosine_similarity(state_vec, t.state_vec) >= self._similarity_threshold
        )
        return min(matches / self._confidence_threshold, 1.0)

    def _cosine_similarity(self, a: np.ndarray, b: np.ndarray) -> float:
        norm_a = np.linalg.norm(a)
        norm_b = np.linalg.norm(b)
        if norm_a == 0 or norm_b == 0:
            return 1.0 if norm_a == 0 and norm_b == 0 else 0.0
        return float(np.dot(a, b) / (norm_a * norm_b))
```

```python
# arc_see/agent.py
"""Core agent loop: observe → perceive → consult → act → record."""

from __future__ import annotations

import numpy as np
import random

from arc_see.composer import Composer
from arc_see.graph import MemoryGraph
from arc_see.lenses import Annotation, RegionLabels, EdgeMask, GoalDetection


class Agent:
    def __init__(self):
        self.composer = Composer()
        self.graph = MemoryGraph()
        self.history: list[np.ndarray] = []
        self._annotations: dict[str, Annotation] = {}
        self._prev_annotations: dict[str, Annotation] = {}
        self._prev_state_vec: np.ndarray | None = None
        self._exploration_actions: list[str] = [
            "ACTION1", "ACTION2", "ACTION3", "ACTION4", "ACTION5",
        ]
        self._exploration_idx: int = 0

    def observe(self, grid: np.ndarray) -> None:
        """Store a new frame."""
        self.history.append(grid.copy())

    def perceive(self) -> dict[str, Annotation]:
        """Run all lenses on current frame."""
        if not self.history:
            return {}
        self._prev_annotations = self._annotations
        self._annotations = self.composer.perceive(
            self.history[-1],
            self.history[:-1],
        )
        return self._annotations

    def select_action(
        self,
        available_actions: list[str],
    ) -> tuple[str, dict]:
        """Pick an action based on graph confidence."""
        state_vec = self._vectorize(self._annotations)
        confidence = self.graph.confidence(state_vec)

        if confidence > 0.5:
            # Exploitation: ask the graph
            results = self.graph.query(state_vec, max_results=5)
            for r in results:
                if r.action in available_actions:
                    data = {}
                    if r.action == "ACTION6":
                        data = self._pick_click_target()
                    return r.action, data

        # Exploration: cycle through simple actions, then click on structure
        return self._explore(available_actions)

    def record(self, action: str, data: dict) -> None:
        """Record the transition in the memory graph."""
        if len(self.history) < 2:
            return

        prev_vec = self._vectorize(self._prev_annotations) if self._prev_annotations else np.zeros(1)
        curr_vec = self._vectorize(self._annotations)

        # Did the state change?
        prev_grid = self.history[-2] if len(self.history) >= 2 else self.history[-1]
        curr_grid = self.history[-1]
        state_changed = not np.array_equal(prev_grid, curr_grid)

        # Goal delta
        goal_delta = self._goal_delta()

        # Active lens combinations
        top = self.composer.top_k(self._annotations, k=3)
        active_lenses = [pair for _, pair in top]

        self.graph.record(
            state_vec=prev_vec,
            action=action,
            next_state_vec=curr_vec,
            state_changed=state_changed,
            goal_delta=goal_delta,
            active_lenses=active_lenses,
        )

    def _explore(self, available_actions: list[str]) -> tuple[str, dict]:
        """Exploration strategy: cycle simple actions, then structured clicks."""
        # First exhaust simple actions
        while self._exploration_idx < len(self._exploration_actions):
            action = self._exploration_actions[self._exploration_idx]
            self._exploration_idx += 1
            if action in available_actions:
                return action, {}

        # Then click on interesting locations
        if "ACTION6" in available_actions:
            return "ACTION6", self._pick_click_target()

        # Fallback: random available action
        action = random.choice(available_actions)
        data = {}
        if action == "ACTION6":
            data = self._pick_click_target()
        return action, data

    def _pick_click_target(self) -> dict:
        """Pick a click target based on lens annotations."""
        # Prefer clicking on edges, object boundaries, or structure
        if "edges" in self._annotations:
            edge_mask = self._annotations["edges"]
            if isinstance(edge_mask, EdgeMask) and edge_mask.mask.any():
                ys, xs = np.where(edge_mask.mask)
                idx = random.randint(0, len(ys) - 1)
                return {"x": int(xs[idx]), "y": int(ys[idx])}

        # Fallback: random position
        return {"x": random.randint(0, 63), "y": random.randint(0, 63)}

    def _vectorize(self, annotations: dict[str, Annotation]) -> np.ndarray:
        """Convert annotations to a fixed-size vector for graph lookups."""
        features: list[float] = []

        # Region count
        cc = annotations.get("connected_components")
        features.append(float(cc.count) if isinstance(cc, RegionLabels) else 0.0)

        # Edge density
        edges = annotations.get("edges")
        if isinstance(edges, EdgeMask):
            features.append(float(edges.mask.sum()) / (64 * 64))
        else:
            features.append(0.0)

        # Symmetry flags
        sym = annotations.get("symmetry")
        if sym and hasattr(sym, 'horizontal'):
            features.extend([
                float(sym.horizontal),
                float(sym.vertical),
                float(sym.rotational_order) / 4.0,
            ])
        else:
            features.extend([0.0, 0.0, 0.0])

        # Goal similarity
        goal = annotations.get("goal_detection")
        if isinstance(goal, GoalDetection):
            features.append(goal.similarity)
        else:
            features.append(0.0)

        # Periodicity
        per = annotations.get("periodicity")
        if per and hasattr(per, 'row_period'):
            features.append(float(per.row_period or 0) / 64.0)
            features.append(float(per.col_period or 0) / 64.0)
        else:
            features.extend([0.0, 0.0])

        # Movement magnitude
        mov = annotations.get("movement")
        if mov and hasattr(mov, 'vectors') and mov.vectors:
            magnitudes = [
                (dy**2 + dx**2)**0.5 for dy, dx in mov.vectors.values()
            ]
            features.append(float(np.mean(magnitudes)) / 64.0)
        else:
            features.append(0.0)

        # Change density
        change = annotations.get("creation_destruction")
        if change and hasattr(change, 'appeared'):
            features.append(
                float(change.appeared.sum() + change.disappeared.sum()) / (64 * 64)
            )
        else:
            features.append(0.0)

        return np.array(features, dtype=np.float32)

    def _goal_delta(self) -> float:
        """How much did goal similarity change between prev and current?"""
        prev_goal = self._prev_annotations.get("goal_detection")
        curr_goal = self._annotations.get("goal_detection")
        prev_sim = prev_goal.similarity if isinstance(prev_goal, GoalDetection) else 0.0
        curr_sim = curr_goal.similarity if isinstance(curr_goal, GoalDetection) else 0.0
        return curr_sim - prev_sim
```

```python
# arc_see/runner.py
"""ARC-AGI-3 API integration: play games using the agent."""

from __future__ import annotations

import numpy as np

import arc_agi
from arcengine import GameAction, GameState

from arc_see.agent import Agent


# Map string action names to GameAction enum values
_ACTION_MAP = {
    "ACTION1": GameAction.ACTION1,
    "ACTION2": GameAction.ACTION2,
    "ACTION3": GameAction.ACTION3,
    "ACTION4": GameAction.ACTION4,
    "ACTION5": GameAction.ACTION5,
    "ACTION6": GameAction.ACTION6,
    "ACTION7": GameAction.ACTION7,
}

_REVERSE_ACTION_MAP = {v: k for k, v in _ACTION_MAP.items()}


def play_game(
    game_id: str,
    max_turns: int = 500,
    render_mode: str | None = "terminal",
    verbose: bool = True,
) -> dict:
    """Play a single ARC-AGI-3 game using the lens agent.

    Returns a summary dict with turns, levels completed, and lens stats.
    """
    arcade = arc_agi.Arcade()
    env = arcade.make(game_id, render_mode=render_mode)
    if env is None:
        raise RuntimeError(f"Failed to create environment for {game_id}")

    agent = Agent()
    turns = 0
    levels_completed = 0
    lens_usage: dict[tuple[str, str], int] = {}

    # Initial observation
    initial_grid = _extract_grid(env)
    agent.observe(initial_grid)
    agent.perceive()

    for turn in range(max_turns):
        # Available actions as strings
        available = [
            _REVERSE_ACTION_MAP[a]
            for a in env.action_space
            if a in _REVERSE_ACTION_MAP
        ]
        if not available:
            break

        # Select action
        action_name, data = agent.select_action(available)
        game_action = _ACTION_MAP[action_name]

        # Execute
        obs = env.step(game_action, data=data if data else None)
        turns += 1

        if obs is None:
            break

        # Observe result
        grid = np.array(obs.frame[-1], dtype=np.int64) if obs.frame else _extract_grid(env)
        agent.observe(grid)
        agent.perceive()
        agent.record(action_name, data)

        # Track which lens combinations were used
        top = agent.composer.top_k(agent._annotations, k=3)
        for _, pair in top:
            lens_usage[pair] = lens_usage.get(pair, 0) + 1

        if verbose and turn % 50 == 0:
            conf = agent.graph.confidence(agent._vectorize(agent._annotations))
            print(f"Turn {turn}: graph_size={agent.graph.size}, confidence={conf:.2f}")

        # Check game state
        if obs.state == GameState.WIN:
            levels_completed += 1
            if verbose:
                print(f"Level completed at turn {turn}!")
        elif obs.state == GameState.GAME_OVER:
            if verbose:
                print(f"Game over at turn {turn}")
            break

    # Get scorecard
    scorecard = arcade.get_scorecard()

    return {
        "game_id": game_id,
        "turns": turns,
        "levels_completed": levels_completed,
        "graph_size": agent.graph.size,
        "history_length": len(agent.history),
        "top_lens_combinations": sorted(
            lens_usage.items(), key=lambda x: x[1], reverse=True
        )[:10],
        "score": scorecard.score if scorecard else None,
    }


def _extract_grid(env) -> np.ndarray:
    """Extract the current grid from the environment."""
    if hasattr(env, 'current_frame'):
        return np.array(env.current_frame, dtype=np.int64)
    return np.zeros((64, 64), dtype=np.int64)


if __name__ == "__main__":
    import sys
    game_id = sys.argv[1] if len(sys.argv) > 1 else "ls20"
    max_turns = int(sys.argv[2]) if len(sys.argv) > 2 else 500
    result = play_game(game_id, max_turns=max_turns)
    print("\n=== Results ===")
    for k, v in result.items():
        print(f"  {k}: {v}")
```
